-- MM2 Trader v8.3 -> v8.3.1 patch loader
-- Run THIS instead of the v8.3 script. It reads mm2_trader.txt from your executor
-- workspace (run v8.3 once first so that file exists), applies the fixes below,
-- saves the patched file back (so server hops re-run the patched version), and starts it.
--
-- Fixes:
--  1. Inventory reading: 2 tries per player, and the log says WHY it failed
--     (no reply / userId mismatch / no Weapons.Owned).
--  2. Offers: after the bundle is offered it checks the trade, and re-offers anything
--     that didn't land (up to 3 rounds). Before, a dropped offer just stalled the trade.
--  3. Accept: if the deal is good but my accept never registered, it retries the
--     confirm (up to 3 times, 6s apart) instead of sitting there until timeout.
--  4. Safer defaults: no mass-blocking, no auto "add more" chat, and the Telegram
--     token/chat id are removed from the file (put them in mm2_secrets.json).

local FILE = "mm2_trader.txt"
local MARK = "-- PATCHED831"

if not (readfile and writefile and isfile and isfile(FILE)) then
    warn("mm2_trader.txt not found in workspace. Run the v8.3 script once first.")
    return
end

local raw = readfile(FILE)
local inner = raw
do
    local open = "local src = [==["
    local a = raw:find(open, 1, true)
    if a then
        local s = a + #open
        local e = raw:find("]==]", s, true)
        if e then inner = raw:sub(s, e - 1) end
    end
end

local original = inner
local report = {}

local function findOnce(anchor)
    local i, j = inner:find(anchor, 1, true)
    if not i then return nil end
    if inner:find(anchor, j + 1, true) then return nil, "anchor not unique" end
    return i, j
end

local function patch(name, anchor, repl, mode)
    local i, j = findOnce(anchor)
    if not i then
        report[#report + 1] = name .. ":FAILED"
        warn("patch failed: " .. name)
        return
    end
    if mode == "before" then
        inner = inner:sub(1, i - 1) .. repl .. "\n" .. inner:sub(i)
    elseif mode == "after" then
        inner = inner:sub(1, j) .. "\n" .. repl .. inner:sub(j + 1)
    else
        inner = inner:sub(1, i - 1) .. repl .. inner:sub(j + 1)
    end
    report[#report + 1] = name .. ":ok"
end

if inner:find(MARK, 1, true) then
    print("[patch] already patched, starting")
else
    -- 1) safer defaults + secrets out of the file
    patch("defaults", "local BASE = {", [===[
C.BLOCK_NON_FRIENDS = false
C.CHAT_ADD = false
C.TG_TOKEN = ""
C.TG_CHAT = ""
C._patches = "__PATCHES__"
]===], "before")

    -- 2) inventory reading with retries and reasons
    patch("inventory", "local invCache = {}", [===[
readInventory = function(plr)
    local reason = "?"
    for attemptNo = 1, 2 do
        local res = invoke(invRemote, plr.Name)
        if type(res) ~= "table" then
            reason = "no reply"
        elseif res.userId and res.userId ~= plr.UserId then
            reason = "userId mismatch"
        else
            local owned = type(res.Weapons) == "table" and res.Weapons.Owned
            if type(owned) == "table" then return owned end
            reason = "no Weapons.Owned"
        end
        task.wait(0.6)
    end
    log("inventory unreadable:", plr.Name, "-", reason)
    return nil
end
]===], "before")

    -- 3) verify the bundle landed, re-offer anything missing
    patch("reoffer", "state.bundleDone = os.clock()", [===[
        state._accRetry, state._accSince = 0, nil
        for _round = 1, 3 do
            if not state.inTrade or state.upfront then break end
            local landed = false
            local t0 = os.clock()
            while os.clock() - t0 < 3 and state.inTrade do
                if state.latest then
                    local m = readOffers(state.latest)
                    if sameMultiset(m, state.bundle or {}) then landed = true break end
                end
                task.wait(0.25)
            end
            if landed then break end
            local have = {}
            if state.latest then
                for _, x in ipairs((readOffers(state.latest))) do have[x] = (have[x] or 0) + 1 end
            end
            state.offering = true
            for _, id in ipairs(state.bundle or {}) do
                if not state.inTrade then break end
                if (have[id] or 0) > 0 then
                    have[id] -= 1
                else
                    R.Offer:FireServer(id, "Weapons")
                    log("re-offered", id)
                    task.wait(0.7)
                end
            end
            state.offering = false
        end
]===], "after")

    -- 4) retry confirm if a good deal never got accepted
    patch("reconfirm", "if plr.Parent ~= Players then break end", [===[
        do
            local okAcc = state.inTrade and not state.myAccepted and not state.confirming
                and not state.syncing and not state.offering and #state.curTheirs > 0
                and sameMultiset(state.curMine, state.bundle or {})
                and state.curTheirV >= state.curMineV * C.MIN_GAIN
                and state.curTheirV - state.curMineV >= C.MIN_PROFIT
            if okAcc then
                state._accSince = state._accSince or os.clock()
                if os.clock() - state._accSince > 6 and (state._accRetry or 0) < 3 then
                    state._accRetry = (state._accRetry or 0) + 1
                    state._accSince = nil
                    log("good deal but not accepted - retrying confirm", state._accRetry)
                    state.lastSig = ""
                    task.spawn(evaluate)
                end
            else
                state._accSince = nil
            end
        end
]===], "after")

    -- 5) show patch status in the start log
    patch("startlog", 'log("v8.3 | only:",', 'log("v8.3.1 patches:", tostring(C._patches), "| only:",', "replace")

    local i, j = inner:find("__PATCHES__", 1, true)
    if i then
        inner = inner:sub(1, i - 1) .. table.concat(report, ",") .. inner:sub(j + 1)
    end
    inner = MARK .. "\n" .. inner
    print("[patch] " .. table.concat(report, ", "))
end

local fn, err = loadstring(inner)
if not fn then
    warn("patched script failed to compile: " .. tostring(err) .. " - running unpatched")
    fn, err = loadstring(original)
    if not fn then warn("original failed too: " .. tostring(err)) return end
else
    pcall(writefile, FILE, inner)
end
fn()
