local function loadJson(path)
    if path ~= "" and isfile and readfile and isfile(path) then
        local ok, t = pcall(function() return HttpService:JSONDecode(readfile(path)) end)
        if ok and type(t) == "table" then return t end
    end
    return {}
end
local function saveJson(path, t)
    if path ~= "" and writefile then
        pcall(function() writefile(path, HttpService:JSONEncode(t)) end)
    end
end
local function httpGet(url)
    local ok, body = pcall(function() return game:HttpGet(url) end)
    if ok and body then return body end
    if httpReq then
        local ok2, res = pcall(httpReq, { Url = url, Method = "GET" })
        if ok2 and res and res.Body then return res.Body end
    end
    return nil
end

local function log(...)
    local parts = {}
    for _, v in ipairs({...}) do parts[#parts + 1] = tostring(v) end
    local s = table.concat(parts, " ")
    if #s > 200 then s = s:sub(1, 200) .. "..." end
    print("[MM2Trader]", s)
    lines[#lines + 1] = s
    if #lines > 250 then table.remove(lines, 1) end
    if label.Parent then
        label.Text = table.concat(lines, "\n")
        scroll.CanvasPosition = Vector2.new(0, 1e6)
    end
    if C.LOG_FILE ~= "" and appendfile then
        pcall(appendfile, C.LOG_FILE, os.date("%H:%M:%S ") .. s .. "\n")
    end
    refreshStatus()
end

local function dump(v, d)
    d = d or 0
    if d > 2 then return "..." end
    if typeof(v) == "table" then
        local out, n = {}, 0
        for k, x in pairs(v) do
            n += 1
            if n > 8 then out[#out + 1] = "..." break end
            out[#out + 1] = tostring(k) .. "=" .. dump(x, d + 1)
        end
        return "{" .. table.concat(out, ",") .. "}"
    end
    return tostring(v)
end

local function shuffle(t)
    for i = #t, 2, -1 do
        local j = math.random(i)
        t[i], t[j] = t[j], t[i]
    end
    return t
end

local function norm(s) return (tostring(s):lower():gsub("[^%w]", "")) end
local V, DEM = {}, {}
local function mergeInto(t)
    for k, v in pairs(t) do
        if type(v) == "number" then
            V[norm(k)] = v
        elseif type(v) == "table" then
            if type(v.v) == "number" then V[norm(k)] = v.v end
            if type(v.d) == "number" then DEM[norm(k)] = v.d end
        end
    end
end
local function loadValues()
    V, DEM = {}, {}
    for d, names in pairs(DEMAND_TIERS) do
        for _, n in ipairs(names) do DEM[norm(n)] = d end
    end
    mergeInto(BASE)
    mergeInto(C.CUSTOM_VALUES)
    if C.VALUES_URL ~= "" then
        local body = httpGet(C.VALUES_URL)
        if body then
            local ok, t = pcall(function() return HttpService:JSONDecode(body) end)
            if ok and type(t) == "table" then mergeInto(t) log("values: url loaded") else log("values: url bad json") end
        else
            log("values: url fetch failed")
        end
    end
    mergeInto(loadJson(C.VALUES_FILE))
end

local ALIASES = {
    ["Scythe"] = "Batwing",
    ["ElderwoodKnife"] = "Elderwood Blade",
    ["ElderwoodGun"] = "Elderwood Revolver",
}
local function dem(name)
    name = tostring(name)
    local d = DEM[norm(ALIASES[name] or name)]
    if d then return d end
    local stripped = name:gsub("_[KG]_%d+$", ""):gsub("_[KG]$", "")
    return DEM[norm(ALIASES[stripped] or stripped)] or C.DEMAND_DEFAULT
end
local function val(name)
    name = tostring(name)
    local v = V[norm(ALIASES[name] or name)]
    if v then return v end
    local stripped = name:gsub("_[KG]_%d+$", ""):gsub("_[KG]$", "")
    return V[norm(ALIASES[stripped] or stripped)] or 0
end
local function aval(name)
    local raw = val(name)
    if not C.USE_DEMAND or raw == 0 then return raw end
    local d = dem(name)
    if d > 8 then d = 8 end
    return math.floor(raw * (C.DEMAND_MULT[d] or 1) + 0.5)
end

local function displayName(id)
    return tostring(id):gsub("_[KG]_%d+$", ""):gsub("_[KG]$", "")
end

    if #fresh > 0 and who ~= "theirs" then
        log("no value for:", table.concat(fresh, ", "))
        notify("unknown", "Items with no value", table.concat(fresh, ", "))
    end
end

local function invoke(rf, arg)
    local res, done = nil, false
    task.spawn(function()
        local ok, r = pcall(function() return rf:InvokeServer(arg) end)
        if ok then res = r end
        done = true
    end)
    local t = os.clock()
    while not done and os.clock() - t < 6 do task.wait(0.1) end
    return res
end

local keysLogged = false
local function readInventory(plr)
    local reason = "?"
    for attemptNo = 1, 2 do
        local res = invoke(invRemote, plr.Name)
        if type(res) ~= "table" then
            reason = "no reply"
        elseif res.userId and res.userId ~= plr.UserId then
            reason = "userId mismatch"
        else
            local owned = type(res.Weapons) == "table" and res.Weapons.Owned
            if type(owned) == "table" then
                if not keysLogged then
                    keysLogged = true
                    local k = {}
                    for key in pairs(res) do k[#k + 1] = tostring(key) end
                    table.sort(k)
                    local str = table.concat(k, ",")
                    for i = 1, #str, 180 do log("inv keys:", str:sub(i, i + 179)) end
                end
                return owned
            end
            reason = "no Weapons.Owned"
        end
        task.wait(0.6)
    end
    log("inventory unreadable:", plr.Name, "-", reason)
    return nil
end

local invCache = {}
local function cachedInventory(plr)
    local c = invCache[plr.UserId]
    if c and os.clock() - c.t < C.INV_TTL then return c.owned end
    local owned = readInventory(plr)
    if owned then invCache[plr.UserId] = { t = os.clock(), owned = owned } end
    return owned
end

local function offerable(id, v)
    if C.KEEP[id] or C.NEVER_OFFER[id] or v <= 0 then return false end
    if next(C.ONLY_OFFER) ~= nil then return C.ONLY_OFFER[id] == true end
    return v >= C.MIN_ITEM_OFFER and v <= C.MAX_ITEM_OFFER
end

local function isGodly(id)
    if C.GODLY_EXTRA[id] then return true end
    local n = tostring(id)
    if n:find("Chroma") and not C.GODLY_INCLUDE_CHROMA then return false end
    local raw = val(id)
    return raw >= 800
end

local function myItems()
    local owned = readInventory(me)
    if not owned then return nil end
    local list, unknown = {}, {}
    for id, c in pairs(owned) do
        id = tostring(id)
        local raw = val(id)
        if raw == 0 and not C.KEEP[id] then unknown[#unknown + 1] = id end
        if offerable(id, raw) then
            local av = aval(id)
            for _ = 1, math.min(tonumber(c) or 1, C.MAX_OFFER_ITEMS) do
                list[#list + 1] = { id = id, v = av, raw = raw }
            end
        end
    end
    if #unknown > 0 then noteUnknown(unknown, "mine") end
    table.sort(list, function(a, b) return a.v > b.v end)
    return list
end

local function buildBundle(list, budget)
    local pool = {}
    for _, it in ipairs(list) do
        if it.v <= budget then
            pool[#pool + 1] = it
            if #pool >= C.POOL_LIMIT then break end
        end
    end
    local best, bestTot = {}, 0
    local picks, nodes = {}, 0
    local function dfs(i, tot)
        if tot > bestTot or (tot == bestTot and tot > 0 and #picks < #best) then
            best = table.clone(picks)
            bestTot = tot
        end
        if #picks >= C.MAX_OFFER_ITEMS or bestTot >= budget then return end
        for j = i, #pool do
            nodes += 1
            if nodes > 20000 then return end
            local v = pool[j].v
            if tot + v <= budget then
                picks[#picks + 1] = pool[j].id
                dfs(j + 1, tot + v)
                picks[#picks] = nil
                if bestTot >= budget then return end
            end
        end
    end
    dfs(1, 0)
    return best, bestTot
end

local function pickDeal(theirOwned, mylist)
    local cands, unk = {}, {}
    for id in pairs(theirOwned) do
        id = tostring(id)
        local raw = val(id)
        if raw == 0 then unk[#unk + 1] = id end
        local d = dem(id)
        if raw >= C.MIN_TARGET_VALUE and raw <= C.MAX_TARGET_VALUE
            and (not C.USE_DEMAND or d >= C.MIN_TARGET_DEMAND) then
            cands[#cands + 1] = { id = id, v = aval(id), raw = raw, d = d }
        end
    end
    if #unk > 0 then noteUnknown(unk, "theirs") end
    table.sort(cands, function(a, b) return a.v > b.v end)
    local best
    for i = 1, math.min(#cands, C.CANDIDATES) do
        local vT = cands[i].v
        local budget = math.min(vT / C.MIN_GAIN, C.MAX_TRADE_RISK)
        local bundle, tot = buildBundle(mylist, budget)
        if #bundle > 0 and tot >= budget * C.MIN_FILL and vT - tot >= C.MIN_PROFIT then
            local score = vT - tot
            if C.DEMAND_BIAS and C.DEMAND_BIAS > 0 then
                score = score * (1 + C.DEMAND_BIAS * (cands[i].d - 4))
            end
            if C.LIQUIDITY_GUARD then
                local theirD = cands[i].d
                local myMinD = 99
                for _, id in ipairs(bundle) do
                    myMinD = math.min(myMinD, dem(id))
                end
                if theirD < C.LIQUIDITY_MIN_D and myMinD >= C.LIQUIDITY_MIN_D then
                    -- skip
                else
                    if not best or score > best.score then
                        best = { item = cands[i].id, vT = vT, raw = cands[i].raw, d = cands[i].d,
                                 bundle = bundle, tot = tot, score = score }
                    end
                end
            else
                if not best or score > best.score then
                    best = { item = cands[i].id, vT = vT, raw = cands[i].raw, d = cands[i].d,
                             bundle = bundle, tot = tot, score = score }
                end
            end
        end
    end
    return best
end

local function godlyPool(mylist)
    local g = {}
    for _, it in ipairs(mylist) do
        if isGodly(it.id) then g[#g + 1] = it end
    end
    return g
end

local function makeDeal(theirOwned, mylist)
    local deal = pickDeal(theirOwned, mylist)
    if not deal then return nil end
    if C.UPFRONT_GODLYS and R.Remove and not removeBroken then
        local pool = godlyPool(mylist)
        if #pool > 0 then
            deal.upfront = true
            deal.pool = pool
        end
    end
    return deal
end

local addCount, lastAdd = 0, 0
local function promptAdd(kind)
    if not C.CHAT_ADD or addCount >= C.MAX_ADD_PROMPTS or os.clock() - lastAdd < C.ADD_COOLDOWN then return end
    local pool, msg
    if C.CHAT_NAME_TARGET and state.targetName then
        pool = C.ADD_NAMED
        local tmpl = pool[math.random(#pool)]
        msg = tmpl:gsub("{item}", state.targetName)
    else
        pool = (kind == "fair") and C.ADD_FAIR or C.ADD_LOW
        msg = pool[math.random(#pool)]
    end
    if say(msg) then
        addCount += 1
        lastAdd = os.clock()
    end
end

local function isMe(p) return p == me or tostring(p) == me.Name end

local function readOffers(data)
    local mine, theirs, other, myAcc = {}, {}, nil, false
    if type(data) ~= "table" then return mine, theirs, other, myAcc end
    for _, key in ipairs({"Player1", "Player2"}) do
        local side = data[key]
        if type(side) == "table" then
            local mineSide = isMe(side.Player)
            if mineSide then myAcc = side.Accepted == true else other = tostring(side.Player) end
            local into = mineSide and mine or theirs
            for _, it in pairs(side.Offer or {}) do
                if type(it) == "table" and it[1] then
                    for _ = 1, (tonumber(it[2]) or 1) do table.insert(into, tostring(it[1])) end
                end
            end
        end
    end
    return mine, theirs, other, myAcc
end

local function total(list) local t = 0 for _, n in ipairs(list) do t += aval(n) end return t end
local function rawTotal(list) local t = 0 for _, n in ipairs(list) do t += val(n) end return t end

local function sameMultiset(a, b)
    if #a ~= #b then return false end
    local c = {}
    for _, x in ipairs(a) do c[x] = (c[x] or 0) + 1 end
    for _, x in ipairs(b) do
        if not c[x] or c[x] == 0 then return false end
        c[x] -= 1
    end
    return true
end

local function resetTradeState()
    state.inTrade, state.completed, state.latest, state.lastSig, state.myAccepted = false, false, nil, "", false
    state.confirming, state.theirCount, state.bundleDone, state.nudged, state.recorded = false, 0, nil, false, false
    state.curMineV, state.curTheirV, state.curMineR, state.curTheirR = 0, 0, 0, 0
    state.curMine, state.curTheirs = {}, {}
    state.syncing, state.syncCount, state.offering = false, 0, false
    state._accRetry, state._accSince = 0, nil
    state.theirSig, state.theirStable = "", 0
    addCount, lastAdd = 0, 0
end

    R.Start.OnClientEvent:Connect(function(data, other)
        local otherName = tostring(other)
        if not state.target or otherName:lower() ~= state.target:lower() then
            log("unexpected trade with", otherName, "- doing nothing")
            return
        end
        resetTradeState()
        state.inTrade = true
        log("trade started with", otherName)
        task.wait(1)
        state.offering = true
        local toOffer = state.upfront and initialGodlys(state.pool) or (state.bundle or {})
        for _, id in ipairs(toOffer) do
            if not state.inTrade then break end
            R.Offer:FireServer(id, "Weapons")
            log("offered", id, "(" .. val(id) .. ", d" .. dem(id) .. ")")
            task.wait(0.6)
        end
        state.offering = false
        state.bundleDone = os.clock()

        -- v8.3.1: re-offer anything that didn't land
        for _round = 1, 3 do
            if not state.inTrade or state.upfront then break end
            local landed = false
            local t0 = os.clock()
            while os.clock() - t0 < 3 and state.inTrade do
                if state.latest then
                    local m = readOffers(state.latest)
                    if sameMultiset(m, state.bundle or {}) then landed = true break end
                end
                task.wait(0.25)
            end
            if landed then break end
            local have = {}
            if state.latest then
                for _, x in ipairs((readOffers(state.latest))) do have[x] = (have[x] or 0) + 1 end
            end
            state.offering = true
            for _, id in ipairs(state.bundle or {}) do
                if not state.inTrade then break end
                if (have[id] or 0) > 0 then
                    have[id] -= 1
                else
                    R.Offer:FireServer(id, "Weapons")
                    log("re-offered", id)
                    task.wait(0.7)
                end
            end
            state.offering = false
        end
    end)
end

local function confirm()
    local ld = state.latest
    if not (ld and R.Confirm) or state.confirming then return end
    state.confirming = true
    if ld.LastOffer then
        R.Confirm:FireServer(ld.LastOffer)
        log("confirm sent with LASTOFFER")
    else
        R.Confirm:FireServer(570)
        log("confirm sent with 570")
    end
    task.wait(1.5)
    state.confirming = false
end

local function callRemote(r, ...)
    local args = table.pack(...)
    pcall(function()
        if r:IsA("RemoteFunction") then
            r:InvokeServer(table.unpack(args, 1, args.n))
        else
            r:FireServer(table.unpack(args, 1, args.n))
        end
    end)
end

local function initialGodlys(pool)
    local out = {}
    for i = 1, math.min(#pool, C.UPFRONT_MAX) do out[#out + 1] = pool[i].id end
    return out
end

local function desiredBundle(theirs, mine)
    if state.syncCount >= C.MAX_SYNCS then return table.clone(mine) end
    local budget = math.min(total(theirs) / C.MIN_GAIN, C.MAX_TRADE_RISK)
    local bundle, tot = buildBundle(state.pool, budget)
    if #bundle == 0 or tot < budget * C.MIN_FILL then return {} end
    return bundle
end

local function syncOffer(mine, desired)
    state.syncing = true
    state.syncCount += 1
    local keep = {}
    for _, id in ipairs(desired) do keep[id] = (keep[id] or 0) + 1 end
    local toRemove = {}
    for _, id in ipairs(mine) do
        if (keep[id] or 0) > 0 then keep[id] -= 1 else toRemove[#toRemove + 1] = id end
    end
    local toAdd = {}
    for id, n in pairs(keep) do
        for _ = 1, n do toAdd[#toAdd + 1] = id end
    end
    log("adjusting: remove [" .. table.concat(toRemove, ",") .. "] add [" .. table.concat(toAdd, ",") .. "]")
    for _, id in ipairs(toRemove) do
        if not state.inTrade then break end
        if C.REMOVE_WITH_TYPE then callRemote(R.Remove, id, "Weapons") else callRemote(R.Remove, id) end
        task.wait(0.6)
    end
    for _, id in ipairs(toAdd) do
        if not state.inTrade then break end
        R.Offer:FireServer(id, "Weapons")
        task.wait(0.6)
    end
    local ok, t = false, os.clock()
    while state.inTrade and os.clock() - t < 5 do
        local m = readOffers(state.latest)
        if sameMultiset(m, desired) then ok = true break end
        task.wait(0.25)
    end
    state.syncing = false
    if not state.inTrade then return false end
    if not ok then
        removeBroken = true
        state.sysFail = true
        log("remove did not work (wrong remote/args?) - declining, godly-first off until restart")
        notify("error", "Godly-first: remove failed", "Remote: " .. tostring(R.Remove and R.Remove.Name) .. " - check the [OUT] log")
        pcall(function() R.Decline:FireServer() end)
        state.inTrade = false
        return false
    end
    return true
end

local evaluate
local evalDepth = 0
evaluate = function()
    if state.syncing or state.offering then return end
    if evalDepth > 2 then return end
    evalDepth += 1
    local data = state.latest
    if not state.inTrade or type(data) ~= "table" then evalDepth -= 1 return end
    local mine, theirs, other, myAcc = readOffers(data)
    state.myAccepted = myAcc
    if other and state.target and other:lower() ~= state.target:lower() then evalDepth -= 1 return end
    state.theirCount = #theirs
    local theirSig = table.concat(theirs, ",")
    if theirSig ~= state.theirSig then
        state.theirSig = theirSig
        state.theirStable = os.clock()
    end
    local sig = table.concat(mine, ",") .. "|" .. theirSig
    if sig == state.lastSig then evalDepth -= 1 return end
    state.lastSig = sig
    local mv, tv = total(mine), total(theirs)
    local mr, tr = rawTotal(mine), rawTotal(theirs)
    state.curMineV, state.curTheirV, state.curMineR, state.curTheirR = mv, tv, mr, tr
    state.curMine, state.curTheirs = mine, theirs
    log("mine(" .. mv .. "/" .. mr .. "):", table.concat(mine, ","), "| theirs(" .. tv .. "/" .. tr .. "):", table.concat(theirs, ","))
    local zero = {}
    for _, id in ipairs(theirs) do if val(id) == 0 then zero[#zero + 1] = id end end
    for _, id in ipairs(mine) do if val(id) == 0 then zero[#zero + 1] = id end end
    if #zero > 0 then noteUnknown(zero, "trade") end

    if #theirs == 0 then evalDepth -= 1 return end

    -- adaptive offer
    if C.ADAPT_OFFER and R.Remove and not removeBroken and not state.upfront then
        if os.clock() - state.theirStable >= (C.ADAPT_SETTLE or 2) then
            local desired = desiredBundle(theirs, mine)
            if #desired > 0 and not sameMultiset(mine, desired) then
                if not syncOffer(mine, desired) then evalDepth -= 1 return end
                state.lastSig = ""
                evalDepth -= 1
                return evaluate()
            end
        end
    end

    if state.upfront then
        local desired = desiredBundle(theirs, mine)
        if #desired == 0 then
            state.upfront = false
            if #mine > 0 then
                if not syncOffer(mine, {}) then evalDepth -= 1 return end
                state.lastSig = ""
                evalDepth -= 1
                return evaluate()
            end
        else
            state.bundle = desired
            if not sameMultiset(mine, desired) then
                if not syncOffer(mine, desired) then evalDepth -= 1 return end
                state.lastSig = ""
                evalDepth -= 1
                return evaluate()
            end
        end
    end

    if not sameMultiset(mine, state.bundle or {}) then evalDepth -= 1 return end
    local good = (C.TEST_ACCEPT and C.ONLY_PLAYER ~= "" and #theirs > 0)
        or (tv >= mv * C.MIN_GAIN and tv - mv >= C.MIN_PROFIT)
    if not good then
        log("not enough:", tv, "<", math.max(mv * C.MIN_GAIN, mv + C.MIN_PROFIT))
        promptAdd(tv >= mv and "fair" or "low")
        evalDepth -= 1
        return
    end
    task.wait(1.5)
    local m2, t2 = readOffers(state.latest)
    if state.inTrade and table.concat(m2, ",") .. "|" .. table.concat(t2, ",") == sig then
        log("accepting: they give", tv, "for my", mv, "(raw " .. tr .. " / " .. mr .. ")")
        confirm()
    else
        log("offer changed, not confirming")
    end
    evalDepth -= 1
end

local REFUSE_WORDS = { "disabled", "not accepting", "cannot", "can't", "busy", "already", "blocked",
                       "unavailable", "declined", "turned off", "requests off", "not allowed" }
local function requestRefused(res)
    if res == false then return true end
    if type(res) == "string" then
        local l = res:lower()
        for _, w in ipairs(REFUSE_WORDS) do
            if l:find(w, 1, true) then return true end
        end
    end
    return false
end

local function attempt(plr, deal)
    state.target, state.targetPlr, state.bundle = plr.Name, plr, deal.bundle
    state.upfront, state.pool, state.sysFail = deal.upfront == true, deal.pool or {}, false
    state.targetName = displayName(deal.item)
    if state.upfront then state.bundle = initialGodlys(state.pool) end
    resetTradeState()
    S.requests += 1
    log("requesting", plr.Name, "for", deal.item, "(adj " .. deal.vT .. ", raw " .. deal.raw .. ", d" .. deal.d .. ")",
        "offering", table.concat(deal.bundle, "+"), "(adj " .. deal.tot .. ")")
    local sendDone, sendRes = false, nil
    task.spawn(function()
        local ok, res = pcall(function() return R.Send:InvokeServer(plr) end)
        sendRes, sendDone = (ok and res or nil), true
        log("SendRequest ->", ok and dump(res) or ("error: " .. tostring(res)))
    end)
    local t = os.clock()
    while not state.inTrade and os.clock() - t < C.REQUEST_TIMEOUT do
        task.wait(0.25)
        if sendDone and requestRefused(sendRes) then break end
    end
    if not state.inTrade then
        local refused = sendDone and requestRefused(sendRes)
        if R.Cancel then pcall(function() R.Cancel:FireServer() end) end
        state.target, state.targetPlr, state.bundle = nil, nil, nil
        if refused then
            log(plr.Name .. " refused:", dump(sendRes))
            markOff(plr)
            return false, "refused"
        elseif not sendDone then
            log("no response from", plr.Name, "(not penalized)")
            return false, nil
        else
            log("trade did not open with", plr.Name, "-", dump(sendRes))
            return false, nil
        end
    end
    t = os.clock()
    while state.inTrade and os.clock() - t < C.TRADE_TIMEOUT do
        task.wait(0.25)
        if plr.Parent ~= Players then break end

        -- v8.3.1: good deal but my accept never registered -> retry the confirm
        do
            local okAcc = state.inTrade and not state.myAccepted and not state.confirming
                and not state.syncing and not state.offering and #state.curTheirs > 0
                and sameMultiset(state.curMine, state.bundle or {})
                and ((C.TEST_ACCEPT and C.ONLY_PLAYER ~= "")
                    or (state.curTheirV >= state.curMineV * C.MIN_GAIN
                        and state.curTheirV - state.curMineV >= C.MIN_PROFIT))
            if okAcc then
                state._accSince = state._accSince or os.clock()
                if os.clock() - state._accSince > 6 and (state._accRetry or 0) < 3 then
                    state._accRetry = (state._accRetry or 0) + 1
                    state._accSince = nil
                    log("good deal but not accepted - retrying confirm", state._accRetry)
                    state.lastSig = ""
                    task.spawn(evaluate)
                end
            else
                state._accSince = nil
            end
        end

        if state.bundleDone and state.theirCount == 0 then
            local idle = os.clock() - state.bundleDone
            if idle > C.IDLE_NUDGE and not state.nudged then
                state.nudged = true
                promptAdd("low")
            end
            if idle > C.IDLE_DECLINE then
                log("nothing added, declining")
                pcall(function() R.Decline:FireServer() end)
                state.inTrade = false
                break
            end
        end
    end
    if state.inTrade then
        log("trade timed out, declining")
        pcall(function() R.Decline:FireServer() end)
        state.inTrade = false
    end
    task.wait(0.5)
    local ok = state.completed
    if state.sysFail and not ok then
        state.sysFail = false
    elseif ok or (sendDone and not state.sysFail) then
        if not ok then S.declined += 1 end
        recordHist(plr, ok)
    end
    state.target, state.targetPlr, state.bundle = nil, nil, nil
    return ok
end

local hopping = false
local function doTeleport(fn)
    queueScript()
    hopping = true
    S.hops += 1
    L.hops += 1
    saveJson(C.STATS_FILE, L)
    saveJson(C.VISITED_FILE, visited)
    local ok = pcall(fn)
    if not ok then hopping = false end
    task.delay(25, function() hopping = false end)
end

local function fetchServers(cursor)
    local path = "/v1/games/" .. game.PlaceId .. "/servers/Public?sortOrder=Desc&limit=100" ..
        (cursor and ("&cursor=" .. cursor) or "")
    for _, host in ipairs({ "https://games.roblox.com", "https://games.roproxy.com" }) do
        local body = httpGet(host .. path)
        if body then
            local ok, data = pcall(function() return HttpService:JSONDecode(body) end)
            if ok and type(data) == "table" and type(data.data) == "table" then return data end
        end
    end
    return nil
end

local function hop()
    if hopping then return end
    visited["left:" .. game.JobId] = os.time()
    local candidates, cursor, seen = {}, nil, 0
    for _ = 1, 4 do
        local data = fetchServers(cursor)
        if not data then log("hop: server list fetch failed (http blocked?)") break end
        seen += #data.data
        for _, s in ipairs(data.data) do
            if s.id ~= game.JobId and not visited[s.id] and s.playing and s.maxPlayers
                and s.playing >= C.HOP_MIN_PLAYERS and s.playing < s.maxPlayers - 1 then
                candidates[#candidates + 1] = s
            end
        end
        cursor = data.nextPageCursor
        if #candidates >= 10 or not cursor then break end
    end
    if #candidates == 0 then
        log("hop: no candidates (seen " .. seen .. ") - random teleport")
        notify("hop", "Hop (random)", "no server candidates, seen " .. seen)
        doTeleport(function() TeleportService:Teleport(game.PlaceId, me) end)
        return
    end
    local s = shuffle(candidates)[1]
    visited[s.id] = os.time()
    log("hopping to server with", s.playing, "players")
    notify("hop", "Hopping", "to a server with " .. s.playing .. " players")
    doTeleport(function() TeleportService:TeleportToPlaceInstance(game.PlaceId, s.id, me) end)
end

    hopping = false
end)

local nextTry, fails = {}, {}
local skip = {}
for _, n in ipairs(C.SKIP_PLAYERS) do skip[n:lower()] = true end

local function isFriend(plr)
    if not C.SKIP_FRIENDS then return false end
    local ok, r = pcall(function() return me:IsFriendsWith(plr.UserId) end)
    return ok and r
end

local function eligible(plr)
    if plr == me or plr.Parent ~= Players then return false end
    if C.ONLY_PLAYER ~= "" and plr.Name:lower() ~= C.ONLY_PLAYER:lower() then return false end
    if skip[plr.Name:lower()] then return false end
    if (nextTry[plr.Name] or 0) > os.clock() then return false end
    if blacklisted(plr) then return false end
    local h = hist[tostring(plr.UserId)]
    if h and os.time() - (h.t or 0) < (h.off and C.OFF_COOLDOWN or C.REATTEMPT_SECS) then return false end
    return not isFriend(plr)
end

local function scanPlayers(plist)
    local results, pending = {}, 0
    for _, plr in ipairs(plist) do
        pending += 1
        task.spawn(function()
            local owned = cachedInventory(plr)
            if owned then results[#results + 1] = { plr = plr, owned = owned } end
            pending -= 1
        end)
        task.wait(C.SCAN_STAGGER)
    end
    local t = os.clock()
    while pending > 0 and os.clock() - t < 20 do task.wait(0.2) end
    return results
end

local function summaryText()
    local mins = math.floor((os.clock() - S.started) / 60)
    return string.format("Session: %d trades, net %+d (in %d / out %d), %d hops, %dm up\nLifetime: %d trades, net %+d, %d hops",
        S.trades, S.got - S.given, S.got, S.given, S.hops, mins, L.trades, L.got - L.given, L.hops)
end

local blockWorks = nil
local function blockAll()
    if not C.BLOCK_NON_FRIENDS or blockWorks == false then return end
    local n = 0
    for _, plr in ipairs(Players:GetPlayers()) do
        if (L.blocked or 0) >= C.BLOCK_MAX then log("block cap reached, not blocking more") break end
        if plr ~= me and not isFriend(plr) then
            local ok = pcall(function() me:BlockUser(plr.UserId) end)
            if ok then
                n += 1
                L.blocked = (L.blocked or 0) + 1
                blockWorks = true
            elseif blockWorks == nil then
                blockWorks = false
                log("blocking not supported here (BlockUser failed) - disabled")
                break
            end
            task.wait(0.3)
        end
    end
    if n > 0 then
        saveJson(C.STATS_FILE, L)
        log("blocked " .. n .. " players (total " .. (L.blocked or 0) .. ")")
    end
end

local seenPlayers = loadJson(C.SEEN_FILE)
for k, ts in pairs(seenPlayers) do
    if type(ts) ~= "number" or os.time() - ts > 7200 then seenPlayers[k] = nil end
end
local firstCycle = true
local function markSeen()
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= me then seenPlayers[tostring(plr.UserId)] = os.time() end
    end
    saveJson(C.SEEN_FILE, seenPlayers)
end

local function runCycle()
    while paused and not stopped do task.wait(0.5) end
    if stopped then return false end
    if firstCycle then
        firstCycle = false
        task.wait(4)
        local n, old = 0, 0
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= me and not isFriend(plr) then
                n += 1
                if seenPlayers[tostring(plr.UserId)] then old += 1 end
            end
        end
        if n >= 3 and old / n >= C.STALE_RATIO then
            revisit = true
            log(string.format("server looks stale: %d/%d players already seen", old, n))
        end
        markSeen()
    end
    if revisit then
        revisit = false
        log("landed back on a server I already left - hopping again")
        if C.SERVER_HOP then blockAll() hop() end
        task.wait(10)
        return true
    end

    local mylist = myItems()
    if not mylist then log("my inventory unreadable, retrying") task.wait(5) return true end
    if #mylist == 0 then
        log("no offerable items (value >= " .. C.MIN_ITEM_OFFER .. ") - stopping")
        notify("stop", "Out of items", "No tradeable valued items left.\n" .. summaryText())
        return false
    end
    local parts = {}
    for i = 1, math.min(#mylist, 6) do parts[#parts + 1] = mylist[i].id .. "(" .. mylist[i].v .. ")" end
    log("my items:", #mylist, "(adj) |", table.concat(parts, ", "))

    local cands = {}
    for _, plr in ipairs(Players:GetPlayers()) do
        if eligible(plr) then cands[#cands + 1] = plr end
    end
    local scans = scanPlayers(shuffle(cands))
    local deals = {}
    for _, r in ipairs(scans) do
        local d = makeDeal(r.owned, mylist)
        if d then deals[#deals + 1] = { plr = r.plr, owned = r.owned, deal = d } end
    end
    table.sort(deals, function(a, b) return a.deal.score > b.deal.score end)
    log(string.format("scanned %d/%d players, %d deals", #scans, #cands, #deals))

    local didTrade, dirty = false, false
    for _, e in ipairs(deals) do
        if stopped then return false end
        while paused and not stopped do task.wait(0.5) end
        local plr = e.plr
        if plr.Parent == Players then
            local deal = e.deal
            if dirty then deal = makeDeal(e.owned, mylist) end
            if deal then
                nextTry[plr.Name] = os.clock() + C.PLAYER_COOLDOWN
                local done, why = attempt(plr, deal)
                didTrade = true
                invCache[plr.UserId] = nil
                if done then
                    fails[plr.Name] = nil
                    mylist = myItems() or mylist
                    dirty = true
                    if #mylist == 0 then
                        notify("stop", "Out of items", summaryText())
                        return false
                    end
                elseif why == "refused" then
                    nextTry[plr.Name] = os.clock() + C.OFF_COOLDOWN
                    markOff(plr)
                else
                    local f = (fails[plr.Name] or 0) + 1
                    fails[plr.Name] = f
                    nextTry[plr.Name] = os.clock() + math.min(C.PLAYER_COOLDOWN * 2 ^ f, C.MAX_COOLDOWN)
                end
                task.wait(2)
            end
        end
    end
    if not didTrade then
        if C.SERVER_HOP then blockAll() hop() end
        task.wait(10)
    end
    return true
end
